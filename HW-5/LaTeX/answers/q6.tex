\begin{ابجد}
\item از روی داده‌ها، مقادیر تخمینی $\hat{T}(s, a, s')$ را محاسبه می‌کنیم:
\begin{align*}
     & \hat{T}(X, a, X) = 0, \quad &  & \hat{T}(X, a, Y) = \frac{2}{3} = 0.67, \quad &  & \hat{T}(X, a, Z) = \frac{1}{3} = 0.33, \\
     & \hat{T}(X, b, X) = 1, \quad &  & \hat{T}(X, b, Y) = 0, \quad                  &  & \hat{T}(X, b, Z) = 0                   \\
    \\
     & \hat{T}(Y, a, X) = 0, \quad &  & \hat{T}(Y, a, Y) = \frac{1}{3} = 0.33, \quad &  & \hat{T}(Y, a, Z) = \frac{2}{3} = 0.67, \\
     & \hat{T}(Y, b, X) = 1, \quad &  & \hat{T}(Y, b, Y) = 0, \quad                  &  & \hat{T}(Y, b, Z) = 0                   \\
    \\
     & \hat{T}(Z, a, X) = 0, \quad &  & \hat{T}(Z, a, Y) = 0, \quad                  &  & \hat{T}(Z, a, Z) = 1,                  \\
     & \hat{T}(Z, b, X) = 1, \quad &  & \hat{T}(Z, b, Y) = 0, \quad                  &  & \hat{T}(Z, b, Z) = 0
\end{align*}

\item مقدار تخمینی پاداش‌ها را محاسبه می‌کنیم:
\begin{align*}
     & \hat{R}(X, a, Y) = 3, \quad &  & \hat{R}(X, a, Z) = 3, \quad & \hat{R}(X, b, X) = 1 \\
     & \hat{R}(Y, a, Z) = 4, \quad &  & \hat{R}(Y, a, Y) = 4, \quad & \hat{R}(Y, b, X) = 2 \\
     & \hat{R}(Z, a, Z) = 0, \quad &  & \hat{R}(Z, b, X) = 5
\end{align*}
برای حالت‌هایی که در داده‌ها موجود نیستند، تخمینی برای میزان پاداش نداریم. (البته چون در مدل ساخته شده احتمال انتقال آن حالت‌ها صفر در نظر گرفته شده، پس هیچ‌گاه رخ نمی‌دهند و مقدار پاداش آنها اهمیتی ندارد)

\item در بروزرسانی \lr{Value Iteration}، ارزش جدید هر حالت را طبق رابطه‌ی زیر محاسبه می‌کنیم:
\begin{align*}
    V_{new}(s) = \max_a \sum_{s', r'} p(s', r \mid s,a) [r + \gamma V(s')]
\end{align*}
همچنین برای هر حالت، اکشنی که ارزش بیشتر دارد را به عنوان کنش حریصانه انتخاب می‌کنیم:
\begin{align*}
    V_{1}(X) & = \max \{ \frac{2}{3}(3 + 0.9(0)) + \frac{1}{3}(3 + 0.9(0)), 1(1 + 0.9(0)) \} = 3 &  & \Rightarrow \pi(X) = a \\[1em]
    %
    V_{1}(Y) & = \max \{ \frac{2}{3}(4 + 0.9(0)) + \frac{1}{3}(4 + 0.9(0)), 1(2 + 0.9(0)) \} = 4 &  & \Rightarrow \pi(Y) = a \\[1em]
    %
    V_{1}(Z) & = \max \{ 1(0 + 0.9(0)), 1(5 + 0.9(0)) \} = 5                                     &  & \Rightarrow \pi(Z) = b
\end{align*}

\item \textbf{بله}؛ چون این \lr{MDP} متناهی است و همچنین ضریب تخفیف $\gamma < 1$ دارد. الگوریتم \lr{Value Iteration} تحت این شرایط همگرا می‌شود. (وجود حالت \lr{absorbing} خللی به این موضوع وارد نمی‌کند.)

\end{ابجد}
